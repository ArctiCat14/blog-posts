# 1 Statement

- 给定一个 $n\times m$ 的网格，每个格子有一个互不相同的高度。
- 两位玩家交替移动一面旗帜，每次必须把旗帜移到一个四相邻且高度严格更高的格子，无路可走的玩家输。
- 给 $q$ 个独立询问，每个询问指定起始格子，判断先手胜还是后手胜。
- $1\leq n,m\leq 10^5, n\cdot m\leq 10^5,1\leq h_{i,j} \leq10^9,1\leq q\leq 10^5$

---
# 2 Solution

- 参看 [[组合游戏]]。
- 旗帜只能往更高的格子走，所有可能的移动构成一个有向无环图。
- 存在至少一个严格更高的相邻格子且该相邻格子是必败的则必胜；否则必败。
- 依次处理每个格子，检查其四个邻居中是否有严格更高且必败的，有则该格必胜，否则必败。

```cpp
int dx[4] = {0, 0, -1, 1};
int dy[4] = {-1, 1, 0, 0};

struct node {
    int x, y, z;
    node() {}
    node(int x, int y, int z) : x(x), y(y), z(z) {}
};

void init() {
    
}

void solve(int cas) {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> lose(n, vector<int>(m, -1));
    vector<vector<int>> height(n, vector<int>(m));
    vector<node> vs;
    vs.reserve(n * m);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> height[i][j];
            vs.emplace_back(i, j, height[i][j]);
        }
    }
    sort(all(vs), [](node a, node b) {
        return a.z > b.z;
    });
    for (int i = 0; i < vs.size(); i++) {
        int x = vs[i].x;
        int y = vs[i].y;
        int z = vs[i].z;
        for (int i = 0; i < 4; i++) {
            int ddx = x + dx[i];
            int ddy = y + dy[i];
            if (ddx >= 0 && ddx < n && ddy >= 0 && ddy < m && z < height[ddx][ddy]) {
                if (lose[ddx][ddy] == 1) {
                    lose[x][y] = 0;
                    break;
                }
            }
        }
        if (lose[x][y] == -1) lose[x][y] = 1;
    }
    int q;
    cin >> q;
    for (int i = 0; i < q; i++) {
        int r, c;
        cin >> r >> c;
        r -= 1;
        c -= 1;
        if (lose[r][c] == 1) {
            cout << "Second\n";
        }
        else {
            cout << "First\n";
        }
    }
}
```
