# 1 Statement

- 给定一个 $n\times m$ 的矩阵，每个坐标含一个正整数
- 若坐标 $(r_1, c_1), (r_2,c_2)$ 满足 $r_1<r_2,c_1<c_2$ 且这两个坐标上的数字相等，则矩形 $[r_1,c_1]\times[r_2,c_2]$ 上的所有格子被标记
- 输出整个矩阵最终的标记情况
- $1\leq n\times m\leq 10^6,1\leq a_{i,j} \leq n\times m$

---
# 2 Solution

- 对于矩阵上的每个数 $x$，我们记录它每行的位置 $r_i$，以及对应每行中最小列位置 $\mathrm{mn}_i$ 和最大列位置 $\mathrm{mx}_i$，$(1\leq i \leq k)$。
- 接着我们规定前缀最小列位置 $L_i=\min_{j=1}^{i}\mathrm{mn}_j$，后缀最大列位置 $R_i=\max_{j=i}^{k}\mathrm{mx}_j$。
- 对于 $i\in[1, k-1]$，我们考虑 $r_i$ 和它下一个邻接位置 $r_{i+1}$，若 $L_i<R_{i+1}$，我们就标记矩形
$$[r_i, L_{i}]\times[r_{i+1}, R_{i+1}]$$
- 标记的方法为构造 [[difference-array#2 二维差分数组|二维差分数组]]再求前缀和。

```cpp
void solve() {
    int n, m;
    cin >> n >> m;
    vector<vector<array<int, 3>>> mp(n * m + 1);

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            int x;
            cin >> x;
            if (!sz(mp[x]) || mp[x].back()[0] != i) {
                mp[x].push_back({i, j, j});
            }
            else {
                mp[x].back()[2] = j;
            }
        }
    }

    vector<vector<int>> grid(n + 2, vector<int>(m + 2));
    for (int x = 1; x <= m * n; x++) {
        if (!sz(mp[x])) continue;

        int k = sz(mp[x]);
        if (k < 2) continue;
        
        vector<int> L(k), R(k);
        L[0] = mp[x].front()[1];
        R[k - 1] = mp[x].back()[2];
        for (int i = 1; i < mp[x].size(); i++) {
            L[i] = min(L[i - 1], mp[x][i][1]);
        }
        for (int i = k - 2; i >= 0; i--) {
            R[i] = max(R[i + 1], mp[x][i][2]);
        }

        for (int i = 0; i < k - 1; i++) {
            if (L[i] >= R[i + 1]) continue;
            int r1 = mp[x][i][0];
            int r2 = mp[x][i + 1][0];
            int c1 = L[i];
            int c2 = R[i + 1];
            grid[r1][c1] += 1;
            grid[r1][c2 + 1] -= 1;
            grid[r2 + 1][c1] -= 1;
            grid[r2 + 1][c2 + 1] += 1;
        }
    }
    
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            grid[i][j] += grid[i - 1][j] + grid[i][j - 1] - grid[i - 1][j - 1];
            if (grid[i][j]) {
                cout << "1";
            }
            else {
                cout << "0";
            }
        }
        cout << '\n';
    }
}
```
